================================================================================
ZOHO CREATOR WORKFLOW - FETCH BOOKINGS BY SUPPORT WORKER
================================================================================

This file contains the Deluge script code to use in Zoho Creator workflows
to fetch bookings based on the selected support worker.

================================================================================
OPTION 1: WORKFLOW ON FORM SUBMISSION
================================================================================

Use this in a workflow that triggers when a user submits a form with a 
support worker selection.

Trigger: On Submit
Action: Custom Function

---CODE START---

// Get the selected support worker value from the form
supportWorkerValue = input.Support_worker2;

// Fetch all bookings for this support worker
bookingsList = zoho.creator.getRecords(
    "calender-includa",
    "All_Bookings",
    "Support_worker2 == \"" + supportWorkerValue + "\"",
    1,
    200
);

// Process the fetched bookings
if(bookingsList.size() > 0)
{
    info "Found " + bookingsList.size() + " bookings for support worker: " + supportWorkerValue;
    
    // You can now process each booking
    for each booking in bookingsList
    {
        participantName = booking.get("Participant").get("zc_display_value");
        startDate = booking.get("Start_Date_and_Time");
        endDate = booking.get("End_Date_and_Time");
        status = booking.get("Status");
        
        info "Booking: " + participantName + " from " + startDate + " to " + endDate;
    }
}
else
{
    info "No bookings found for this support worker";
}

---CODE END---

================================================================================
OPTION 2: CUSTOM FUNCTION FOR API/WIDGET INTEGRATION
================================================================================

Use this as a custom function that can be called from your widget or API.

Function Name: fetchBookingsByWorker
Parameters: workerID (text)

---CODE START---

// Function to fetch bookings by support worker ID
Map fetchBookingsByWorker(String workerID)
{
    resultMap = Map();
    bookingsList = List();
    
    try
    {
        // Fetch bookings where Support_worker2 matches the worker ID
        records = zoho.creator.getRecords(
            "calender-includa",
            "All_Bookings",
            "Support_worker2 == \"" + workerID + "\"",
            1,
            200
        );
        
        // Format the results
        for each record in records
        {
            bookingMap = Map();
            bookingMap.put("ID", record.get("ID"));
            bookingMap.put("Participant", record.get("Participant").get("zc_display_value"));
            bookingMap.put("ParticipantID", record.get("Participant").get("ID"));
            bookingMap.put("SupportWorker", record.get("Support_worker2"));
            bookingMap.put("StartDate", record.get("Start_Date_and_Time"));
            bookingMap.put("EndDate", record.get("End_Date_and_Time"));
            bookingMap.put("Status", record.get("Status"));
            bookingMap.put("CRMLink", record.get("CRM_Link2"));
            bookingMap.put("BookingType", record.get("Recurring1"));
            
            bookingsList.add(bookingMap);
        }
        
        resultMap.put("success", true);
        resultMap.put("count", bookingsList.size());
        resultMap.put("bookings", bookingsList);
    }
    catch (e)
    {
        resultMap.put("success", false);
        resultMap.put("error", e.toString());
    }
    
    return resultMap;
}

---CODE END---

================================================================================
OPTION 4: FETCH BY BOTH PARTICIPANT AND SUPPORT WORKER
================================================================================

Fetch bookings filtered by both participant and support worker.

---CODE START---

// Get both filter values
participantID = input.Participant;
supportWorkerValue = input.Support_worker2;

// Build the criteria string
criteria = "";

if(participantID != null && participantID != "")
{
    criteria = "Participant == " + participantID;
}

if(supportWorkerValue != null && supportWorkerValue != "")
{
    if(criteria != "")
    {
        criteria = criteria + " && Support_worker2 == \"" + supportWorkerValue + "\"";
    }
    else
    {
        criteria = "Support_worker2 == \"" + supportWorkerValue + "\"";
    }
}

// Fetch bookings with the combined criteria
if(criteria != "")
{
    bookingsList = zoho.creator.getRecords(
        "calender-includa",
        "All_Bookings",
        criteria,
        1,
        200
    );
    
    info "Found " + bookingsList.size() + " bookings matching the criteria";
}
else
{
    // No filters applied, fetch all
    bookingsList = zoho.creator.getRecords(
        "calender-includa",
        "All_Bookings",
        "",
        1,
        200
    );
    
    info "No filters applied. Fetched all " + bookingsList.size() + " bookings";
}

---CODE END---

================================================================================
OPTION 5: WEBHOOK/API ENDPOINT FOR WIDGET
================================================================================

Create a custom API endpoint that your widget can call to fetch filtered data.

API Function Name: getFilteredBookings
Method: GET
Parameters: participantID (optional), workerID (optional)

---CODE START---

// Get parameters from the API request
participantID = request.getParameter("participantID");
workerID = request.getParameter("workerID");

response = Map();

try
{
    // Build criteria based on provided parameters
    criteria = "";
    
    if(participantID != null && participantID != "")
    {
        criteria = "Participant == " + participantID;
    }
    
    if(workerID != null && workerID != "")
    {
        if(criteria != "")
        {
            criteria = criteria + " && Support_worker2 == \"" + workerID + "\"";
        }
        else
        {
            criteria = "Support_worker2 == \"" + workerID + "\"";
        }
    }
    
    // Fetch the records
    bookingsList = zoho.creator.getRecords(
        "calender-includa",
        "All_Bookings",
        criteria,
        1,
        200
    );
    
    // Format response
    formattedBookings = List();
    
    for each booking in bookingsList
    {
        bookingMap = Map();
        bookingMap.put("id", booking.get("ID"));
        bookingMap.put("participant", booking.get("Participant").get("zc_display_value"));
        bookingMap.put("participantID", booking.get("Participant").get("ID"));
        bookingMap.put("supportWorker", booking.get("Support_worker2"));
        bookingMap.put("startDate", booking.get("Start_Date_and_Time"));
        bookingMap.put("endDate", booking.get("End_Date_and_Time"));
        bookingMap.put("status", booking.get("Status"));
        
        formattedBookings.add(bookingMap);
    }
    
    response.put("success", true);
    response.put("count", formattedBookings.size());
    response.put("data", formattedBookings);
}
catch (e)
{
    response.put("success", false);
    response.put("error", e.toString());
}

// Return JSON response
return response;

---CODE END---

================================================================================
USAGE NOTES
================================================================================

1. Replace "calender-includa" with your actual app name if different
2. Replace "All_Bookings" with your actual report name if different
3. Field names (Support_worker2, Participant, etc.) should match your form fields
4. The limit is set to 200 records - adjust as needed (max is 200 per call)
5. For more than 200 records, implement pagination using the from parameter

6. To use these in Zoho Creator:
   - Go to your app settings
   - Navigate to Workflows or Custom Functions
   - Create a new workflow/function
   - Paste the appropriate code
   - Configure the trigger conditions
   - Save and test

7. For API endpoints:
   - Go to Settings > API
   - Create a new API
   - Add the function code
   - Generate API key if needed
   - Use the endpoint URL in your widget

================================================================================
INTEGRATION WITH CURRENT WIDGET
================================================================================

The current widget.js already handles filtering on the client side.
If you want to move filtering to the server side (Zoho Creator), you would:

1. Create an API endpoint using Option 5 above
2. Modify the loadBookings() function in widget.js to call this API
3. Pass the filter parameters as URL parameters
4. Process the returned JSON data

This would reduce the amount of data transferred and improve performance
for large datasets.
